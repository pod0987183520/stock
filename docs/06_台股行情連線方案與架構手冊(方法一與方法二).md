## 🏆 方法一：Google 試算表 API 結合前端擬真跳動 (當前採用實作)

### 1. 設計哲學與優勢
- **極致零維護**：不依賴任何易失效的第三方網頁爬蟲，由 Google 官方伺服器與交易所介接行情。
- **100% 規避反爬與 CORS**：Google Apps Script (GAS) Web App 原生支援 CORS，絕無驗證碼 (CAPTCHA) 或 IP 封鎖風險。
- **符合台股升降法規**：前端嚴格遵循台灣證券交易所（TWSE）官方 6 大價格升降單位（Tick Size）進行 ±2 單位自然跳動，搭配紅漲綠跌閃爍，提供長輩最真實的看盤沉浸感與防失智大腦刺激。

### 2. 階段一：Google 端建置步驟 (5 分鐘完成)

#### 步驟 1：建立 Google 試算表
1. 開啟 [Google 雲端硬碟](https://drive.google.com/)，新增一個 Google 試算表（命名為 `小股同學_股票行情庫`）。
2. 在試算表工作表 `Sheet1` 中建立以下格式：
   - `A1`: 股票代號（例如 `2330`、`2317`、`2412`、`0050`、`6547`）
   - `B1`: 股票名稱（例如 `台積電`、`鴻海`、`中華電`）
   - `C1`: 填入 Google Finance 公式：`=GOOGLEFINANCE("TPE:" & A1, "price")`（上櫃請用 `"TWO:" & A1`）
   - `D1`: 昨收價公式：`=GOOGLEFINANCE("TPE:" & A1, "closeyest")`

#### 步驟 2：建立 Apps Script Web API
1. 點擊試算表頂部選單【擴充功能】➔【Apps Script】。
2. 將編輯器內的原有代碼替換為以下程式碼：

```javascript
// ==========================================================
// 小股同學 - 官方雲端行情庫 ＋ 自動收錄 ＋ 智慧防擾 ＋ 7天熱門度統計
// ==========================================================

var NOTIFY_EMAIL = "your_email@gmail.com"; // 🔔 請改成您的 Gmail 信箱
var AUTO_ADD_TO_SHEET = true;              // ✅ 是否允許系統自動將新股票加入主表（預設開啟）
var MAX_AUTO_STOCKS = 150;                  // 🛡️ 主表股票數量上限（防止超過 Google 配額）

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var reqCode = (e && e.parameter && e.parameter.code) ? e.parameter.code.toString().trim() : '';

  // 1. 若為單檔代號查詢 (如 ?code=2603)
  if (reqCode) {
    var isBuiltIn = checkIfStockInMainSheet(sheet, reqCode);
    var quote = fetchSingleQuoteDynamic(ss, reqCode);

    // 異步統計與自動收錄 (不阻礙報價回傳速度)
    if (quote && quote.price > 0) {
      try {
        recordStockHit(ss, reqCode);
        if (!isBuiltIn) {
          handleNewStockDiscovered(ss, sheet, reqCode, quote);
        }
      } catch (err) {
        console.warn("背景統計/收錄略過:", err);
      }
    }

    return ContentService.createTextOutput(JSON.stringify(quote || { id: reqCode, price: 0, prevClose: 0 }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // 2. 若無指定參數，全量輸出 Sheet1 內建股票
  var data = sheet.getDataRange().getValues();
  var result = {};
  for (var i = 0; i < data.length; i++) {
    var code = data[i][0] ? data[i][0].toString().trim() : '';
    if (code && code !== '代號') {
      var name = data[i][1] ? data[i][1].toString().trim() : '';
      var price = parseFloat(data[i][2]) || 0;
      var prevClose = parseFloat(data[i][3]) || price;
      result[code] = {
        id: code,
        name: name,
        price: price,
        prevClose: prevClose,
        updatedAt: new Date().toISOString()
      };
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// 現場動態向 Google Finance 查詢單檔即時盤價
function fetchSingleQuoteDynamic(ss, code) {
  var querySheet = ss.getSheetByName("TempQuery") || ss.insertSheet("TempQuery");
  querySheet.hideSheet();
  
  // 先查上市 (TPE)
  querySheet.getRange("A1").setFormula('=GOOGLEFINANCE("TPE:' + code + '", "price")');
  querySheet.getRange("B1").setFormula('=GOOGLEFINANCE("TPE:' + code + '", "name")');
  querySheet.getRange("C1").setFormula('=GOOGLEFINANCE("TPE:' + code + '", "closeyest")');
  SpreadsheetApp.flush();
  
  var price = parseFloat(querySheet.getRange("A1").getValue()) || 0;
  var name = querySheet.getRange("B1").getValue() || ('股票(' + code + ')');
  var prevClose = parseFloat(querySheet.getRange("C1").getValue()) || price;
  var marketType = "TPE";

  // 若查無數值，自動嘗試上櫃 (TWO)
  if (price === 0 || isNaN(price)) {
    querySheet.getRange("A1").setFormula('=GOOGLEFINANCE("TWO:' + code + '", "price")');
    querySheet.getRange("B1").setFormula('=GOOGLEFINANCE("TWO:' + code + '", "name")');
    querySheet.getRange("C1").setFormula('=GOOGLEFINANCE("TWO:' + code + '", "closeyest")');
    SpreadsheetApp.flush();
    price = parseFloat(querySheet.getRange("A1").getValue()) || 0;
    name = querySheet.getRange("B1").getValue() || name;
    prevClose = parseFloat(querySheet.getRange("C1").getValue()) || price;
    marketType = "TWO";
  }

  return {
    id: code,
    name: name,
    price: price,
    prevClose: prevClose,
    marketType: marketType,
    updatedAt: new Date().toISOString()
  };
}

// 檢查是否已在主表
function checkIfStockInMainSheet(sheet, code) {
  var data = sheet.getDataRange().getValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(code)) return true;
  }
  return false;
}

// 處理新發現的有效股票（自動收錄 ＋ 發送 Email）
function handleNewStockDiscovered(ss, sheet, code, quote) {
  var props = PropertiesService.getScriptProperties();
  var todayStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd");
  var notifyKey = "notified_" + code + "_" + todayStr;

  if (props.getProperty(notifyKey)) return; // 今日已處理過，不再重複

  var totalRows = sheet.getLastRow();
  var autoAdded = false;

  // 1. 若開啟自動收錄且未超過上限
  if (AUTO_ADD_TO_SHEET && totalRows < MAX_AUTO_STOCKS) {
    sheet.appendRow([
      code,
      quote.name,
      '=GOOGLEFINANCE("' + quote.marketType + ':' + code + '", "price")',
      '=GOOGLEFINANCE("' + quote.marketType + ':' + code + '", "closeyest")'
    ]);
    autoAdded = true;
  }

  // 2. 發送郵件報告
  try {
    var subject = autoAdded 
      ? "🎉【小股同學】已自動收錄新熱門股：" + code + " " + quote.name
      : "📢【小股同學】收到新股票查詢：" + code + " " + quote.name;
    
    var body = "您好：\n\n有使用者在 App 中查詢/自選了新股票【 " + code + " " + quote.name + " 】，最新報價為 " + quote.price + " 元。\n\n"
             + (autoAdded ? "✅ 系統已全自動將該股票收錄進 Google 試算表，從此享有極速秒開快取！" : "⚠️ 目前主表已達數量上限，請至試算表檢視。")
             + "\n\n目前主表總股票數：" + (autoAdded ? (totalRows + 1) : totalRows) + " 檔"
             + "\n時間：" + new Date().toLocaleString();

    MailApp.sendEmail({ to: NOTIFY_EMAIL, subject: subject, body: body });
    props.setProperty(notifyKey, "true");
  } catch (e) {
    console.warn("郵件發送失敗:", e);
  }
}

// 📊 股票 7 天關注次數自動記錄
function recordStockHit(ss, code) {
  var statSheet = ss.getSheetByName("📊 關注度統計") || ss.insertSheet("📊 關注度統計");
  if (statSheet.getLastRow() === 0) {
    statSheet.appendRow(["股票代號", "累計總查詢", "今日查詢", "前1天", "前2天", "前3天", "前4天", "前5天", "前6天", "前7天", "最後查詢時間", "最後更新日期"]);
  }

  var data = statSheet.getDataRange().getValues();
  var todayStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd");
  var nowStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd HH:mm:ss");
  var rowIdx = -1;

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(code)) {
      rowIdx = i + 1;
      break;
    }
  }

  if (rowIdx > 0) {
    var rowData = data[rowIdx - 1];
    var lastDate = rowData[11] ? Utilities.formatDate(new Date(rowData[11]), "GMT+8", "yyyy-MM-dd") : todayStr;
    var total = (Number(rowData[1]) || 0) + 1;
    var todayHits = (Number(rowData[2]) || 0);

    // 若跨日了，自動將過去數據往後推移，無需手動定時器！
    if (lastDate !== todayStr) {
      statSheet.getRange(rowIdx, 10).setValue(rowData[8]); // 前6天 -> 前7天
      statSheet.getRange(rowIdx, 9).setValue(rowData[7]);
      statSheet.getRange(rowIdx, 8).setValue(rowData[6]);
      statSheet.getRange(rowIdx, 7).setValue(rowData[5]);
      statSheet.getRange(rowIdx, 6).setValue(rowData[4]);
      statSheet.getRange(rowIdx, 5).setValue(rowData[3]);
      statSheet.getRange(rowIdx, 4).setValue(todayHits);    // 今日 -> 前1天
      todayHits = 0;                                       // 今日重置
    }

    statSheet.getRange(rowIdx, 2).setValue(total);
    statSheet.getRange(rowIdx, 3).setValue(todayHits + 1);
    statSheet.getRange(rowIdx, 11).setValue(nowStr);
    statSheet.getRange(rowIdx, 12).setValue(todayStr);
  } else {
    statSheet.appendRow([code, 1, 1, 0, 0, 0, 0, 0, 0, 0, nowStr, todayStr]);
  }
}
```

function getTaiwanTickSize(price) {
  if (price < 10) return 0.01;
  if (price < 50) return 0.05;
  if (price < 100) return 0.1;
  if (price < 500) return 0.5;
  if (price < 1000) return 1.0;

#### 2. 雙週期心跳架構
- **長週期（每 15 分鐘）**：發送 HTTP GET 至 GAS API 網址，更新基準價格（Anchor Base Price）與昨日收盤價。

## 🥈 方法二：富邦新一代 API (Fubon Neo API / WebSocket 轉發)

### 1. 適用場景與特色
- **毫秒級即時撮合**：盤中每一筆真實委託、撮合成交價秒級推播。
- **合規官方通道**：由富邦證券與 Fugle 富果團隊合作開發之現代化 REST/WebSocket API。
- **架構需求**：需常駐運行一台後端伺服器（Node.js / Python）掛載下單數位憑證（`.pfx`），負責與富邦主機建立 WebSocket，再轉發至手機前端。

### 2. 行政申請手續 (100% 線上辦理，免臨櫃)
1. **申請數位交易憑證**：
   - 電腦下載富邦證券官方「憑證 e 總管 (TCEM.exe)」。
   - 登入身分證字號與網頁密碼，點選「申請憑證」，產出 `C:\CAFubon\身分證字號.pfx` 檔案並記住憑證密碼。
2. **簽署 API 風險聲明書**：
   - 登入 [富邦新一代 API 官網](https://www.fbs.com.tw/TradeAPI/) 簽署「API 交易風險預告書」。
3. **完成連線測試**：
   - 下載官方「連線測試小幫手」，完成帳號連線測試即正式啟用。

### 3. 後端轉發伺服器範例 (Node.js + WebSocket Server)

```javascript
// server.js (需在有 Node.js 環境的伺服器執行)
const { FubonSDK } = require('fubon-neo-api');
const WebSocket = require('ws');

const sdk = new FubonSDK();
const wss = new WebSocket.Server({ port: 8080 });

async function initFubon() {
  // 1. 憑證登入
  await sdk.login({
    id: process.env.FUBON_USER_ID,
    pwd: process.env.FUBON_PASSWORD,
    certPath: './cert.pfx',
    certPwd: process.env.FUBON_CERT_PASSWORD
  });
  console.log('✅ 富邦 API 憑證登入成功');

  // 2. 訂閱台積電 (2330) 與 鴻海 (2317) 即時行情
  const ws = sdk.marketdata.websocket();
  ws.on('message', (data) => {
    const quote = JSON.parse(data);
    // 廣播給所有連線的手機前端 App
    wss.clients.forEach(client => {
      if (client.readyState === WebSocket.OPEN) {
        client.send(JSON.stringify({
          id: quote.symbol,
          price: quote.price,
          time: quote.time
        }));
      }
    });
  });
  ws.subscribe(['2330', '2317']);
}

initFubon();
```

## 🥉 方法三：官方合法開放資料源 (TWSE / FinMind 備援)

- **資料端點**：
  - 臺灣證券交易所 OpenAPI：`https://openapi.twse.com.tw/v1/exchangeReport/STOCK_DAY_ALL`

- **特點**：
  - 每日 13:30 收盤後由官方產出當日結算行情。
  - 當【方法一】或【方法二】未設定時，作為系統的自動保底字典庫。
